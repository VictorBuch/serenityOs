** Clipboard
Yank/kill already sync with the macOS clipboard (=select-enable-clipboard= is
~t~ in Doom), so evil ~y~/~p~ talk to it. Cmd is bound to Meta above, so we
can't steal ~M-c/v/x~ without clobbering ~capitalize-word~ / ~M-x~ etc. Instead
use Super, which Option maps to (=mac-option-modifier 'super=): ~Option+c/v/x/a~
become ~s-c/v/x/a~ — free keys, no Meta conflicts. Works in normal, insert, and
visual state plus the minibuffer.

#+begin_src emacs-lisp
(when (eq system-type 'darwin)
  (setq select-enable-clipboard t
        select-enable-primary t)
  ;; Option(=Super)-c/v/x/a copy, paste, cut, select-all.
  (map! :nvi "s-c" #'kill-ring-save     ; Option-c  copy
        :nvi "s-v" #'yank               ; Option-v  paste
        :nvi "s-x" #'kill-region        ; Option-x  cut
        :nvi "s-a" #'mark-whole-buffer) ; Option-a  select all
  (define-key minibuffer-local-map (kbd "s-v") #'yank))
#+end_src

** Emmet (HTML/JSX/TSX/Vue/CSS abbreviations)
Type an abbreviation like ~div.card>ul>li*3~ then expand it. Works in
=web-mode= (so .tsx/.jsx/.vue/.html from the mapping above), plain HTML/SGML,
and CSS-family buffers. =emmet-expand-jsx-className?= makes JSX expand to
~className~ instead of ~class~ — web-mode reports the right content type per
region, so this is safe inside .vue =<template>= too.

Expand with ~C-j~ in insert state (=emmet-expand-line=). After expanding a
template with fields, ~C-j~ / ~C-S-j~ jump between the next/prev edit points.

#+begin_src emacs-lisp
(use-package! emmet-mode
  :hook ((web-mode sgml-mode html-mode mhtml-mode nxml-mode
          css-mode scss-mode less-css-mode) . emmet-mode)
  :config
  (setq emmet-expand-jsx-className? t   ; class -> className in JSX regions
        emmet-self-closing-tag-style " /" ; <br /> not <br/>
        emmet-move-cursor-between-quotes t)
  ;; C-j already maps to emmet-expand-line via emmet-mode-keymap; make it work
  ;; in evil insert state too.
  (map! :map emmet-mode-keymap
        :i "C-j" #'emmet-expand-line))
#+end_src

* Custom
** Session launcher (sesh replacement)
The real replacement for the tmux+sesh flow. ~SPC o s~ pops one fuzzy picker
over *everything you might want to open*:

- ~●~ workspaces already open — switch instantly (your live tmux sessions)
- ~★~ declared projects from =+vb/projects= — switch/create + spawn its servers
- ~z~ zoxide frecency dirs — open any dir you've ~cd~'d into lately
- plain projectile repos — every git project Doom knows about

…and you can just *type a path* that's in none of them. Picking a session
switches/creates its *workspace* (persp-mode), drops you in dired at the root,
and spawns declared servers as background =vterm= buffers (reached with ~SPC ,~).

- ~SPC o s~  — the fuzzy session picker (also on ~SPC p O~)
- ~SPC TAB ]~ / ~SPC TAB [~ — flip between open sessions
- ~SPC TAB 1..9~ — jump to session N (your ctrl+space+<n>)
- ~SPC ,~ — switch buffer *within* this session (the editor + each =*vterm:…*=)

To watch a server: ~SPC ,~ -> pick =*vterm:node*=. To keep it visible while you
edit: ~SPC w v~ (split), then ~SPC ,~ in the new window.

#+begin_src emacs-lisp
(defvar +vb/projects
  '(("web-builder"
     :path "~/Documents/work/web-builder"
     :servers (("docker" . "docker compose up")
               ("node"   . "pnpm run dev")))
    ("serenityOs"
     :path "~/serenityOs"
     :servers nil)
    ("server mal"
     :path "~"
     :shell ("ssh" . "ssh serenity@mal")))
  "Declared session layouts for `+vb/open-session'. Each entry:
NAME plist with :path, optional :servers (alist of NAME . SHELL-CMD run as
background vterms), optional :shell (NAME . CMD for a foreground-only shell
workspace with no editor).")

(defun +vb/vterm-ensure (name dir cmd)
  "Ensure a vterm buffer named *vterm:NAME* exists in DIR running CMD.
CMD is run as the shell's startup command (via `vterm-shell') rather than
typed in afterwards — this avoids the send-string race where input is dropped
before the PTY is ready, leaving an empty buffer. When CMD exits you drop into
a shell so the buffer stays usable. Returns the buffer; never steals the
selected window.

Uses a *login* bash (`bash -lc'): the Emacs daemon is launched by launchd/
systemd with a stripped PATH (only /usr/bin:/bin plus the nix `extraBinPackages'
from doom.nix), so a non-login shell can't find docker, pnpm, etc. A login
shell rebuilds PATH from /etc/profile + path_helper + nix. Bash (not the user's
nushell) is used for reliable POSIX `;'/`exec' semantics."
  (require 'vterm)
  (let ((bufname (format "*vterm:%s*" name)))
    (or (get-buffer bufname)
        (save-window-excursion
          (let* ((default-directory (file-name-as-directory (expand-file-name dir)))
                 (vterm-buffer-name bufname)
                 (sh (or (executable-find "bash") "/bin/bash"))
                 (vterm-shell
                  (if (and cmd (not (string-empty-p cmd)))
                      (format "%s -lc %s" sh
                              (shell-quote-argument (format "%s; exec %s -l" cmd sh)))
                    vterm-shell)))
            (vterm)
            (current-buffer))))))

(defun +vb/--open-dir (name path &optional servers)
  "Switch/create workspace NAME, open dired at PATH, spawn SERVERS as vterms.
The vterms run in the background. They are added to this workspace explicitly
via `persp-add-buffer' — buffers created off-screen (inside
`save-window-excursion') are never displayed in a workspace window, so
persp-mode never adopts them, and `SPC ,' wouldn't list them otherwise.
Reach a running server with `SPC ,'."
  (+workspace-switch name t)
  (dired path)
  (delete-other-windows)
  (dolist (s servers)
    (let ((buf (+vb/vterm-ensure (car s) path (cdr s))))
      (when (and (buffer-live-p buf) (fboundp 'persp-add-buffer))
        (persp-add-buffer buf))))
  (message "Session %s ready%s" name
           (if servers
               (format " — %d server(s) running; SPC , to view" (length servers))
             " — SPC , for its buffers")))

(defun +vb/--session-candidates ()
  "Build DISPLAY -> spec alist from open workspaces, `+vb/projects',
zoxide, and projectile. Deduped by workspace name and by absolute path."
  (let ((seen-name (make-hash-table :test 'equal))
        (seen-path (make-hash-table :test 'equal))
        (out '()))
    (cl-flet ((norm (p) (and p (directory-file-name (expand-file-name p))))
              (add  (display spec)
                (let ((n (plist-get spec :name))
                      (p (and (plist-get spec :path)
                              (directory-file-name
                               (expand-file-name (plist-get spec :path))))))
                  (unless (or (and n (gethash n seen-name))
                              (and p (gethash p seen-path)))
                    (when n (puthash n t seen-name))
                    (when p (puthash p t seen-path))
                    (push (cons display spec) out)))))
      ;; 1. open workspaces — instant switch (running sessions)
      (dolist (w (+workspace-list-names))
        (add (format "● %s" w) (list :name w :kind 'workspace)))
      ;; 2. declared projects
      (dolist (proj +vb/projects)
        (add (format "★ %-14s %s" (car proj) (or (plist-get (cdr proj) :path) ""))
             (append (list :name (car proj) :kind 'config) (cdr proj))))
      ;; 3. zoxide frecency dirs
      (when (executable-find "zoxide")
        (dolist (d (split-string
                    (shell-command-to-string "zoxide query -l") "\n" t))
          (add (format "z %s" (abbreviate-file-name d))
               (list :name (file-name-nondirectory (directory-file-name d))
                     :path d :kind 'dir))))
      ;; 4. projectile known projects
      (dolist (root (and (boundp 'projectile-known-projects)
                         projectile-known-projects))
        (add (format "  %s" (abbreviate-file-name root))
             (list :name (file-name-nondirectory (directory-file-name root))
                   :path root :kind 'dir))))
    (nreverse out)))

(defun +vb/open-session ()
  "Fuzzy session switcher — the sesh replacement. Pick a running workspace,
a declared project, a zoxide dir, a projectile repo, or type any path."
  (interactive)
  (let* ((cands  (+vb/--session-candidates))
         (choice (completing-read "Session: " (mapcar #'car cands) nil nil))
         (spec   (cdr (assoc choice cands))))
    (cond
     ;; free-typed path not in any list
     ((null spec)
      (let ((dir (expand-file-name (string-trim choice))))
        (if (file-directory-p dir)
            (+vb/--open-dir (file-name-nondirectory (directory-file-name dir)) dir)
          (user-error "No session or directory: %s" choice))))
     ;; already-open workspace
     ((eq (plist-get spec :kind) 'workspace)
      (+workspace-switch (plist-get spec :name)))
     ;; declared shell-only session (e.g. ssh mal)
     ((plist-get spec :shell)
      (+workspace-switch (plist-get spec :name) t)
      (switch-to-buffer (+vb/vterm-ensure (car (plist-get spec :shell))
                                          (plist-get spec :path)
                                          (cdr (plist-get spec :shell)))))
     ;; declared project or plain dir
     (t
      (+vb/--open-dir (plist-get spec :name)
                      (expand-file-name (plist-get spec :path))
                      (plist-get spec :servers))))))

(defalias '+vb/open-project #'+vb/open-session)
