** Clipboard
Yank/kill already sync with the macOS clipboard (=select-enable-clipboard= is
~t~ in Doom), so evil ~y~/~p~ talk to it. Cmd is bound to Meta above, so we
can't steal ~M-c/v/x~ without clobbering ~capitalize-word~ / ~M-x~ etc. Instead
use Super, which Option maps to (=mac-option-modifier 'super=): ~Option+c/v/x/a~
become ~s-c/v/x/a~ — free keys, no Meta conflicts. Works in normal, insert, and
visual state plus the minibuffer.

#+begin_src emacs-lisp
(when (eq system-type 'darwin)
  (setq select-enable-clipboard t
        select-enable-primary t)
  ;; Option(=Super)-c/v/x/a copy, paste, cut, select-all.
  (map! :nvi "s-c" #'kill-ring-save     ; Option-c  copy
        :nvi "s-v" #'yank               ; Option-v  paste
        :nvi "s-x" #'kill-region        ; Option-x  cut
        :nvi "s-a" #'mark-whole-buffer) ; Option-a  select all
  (define-key minibuffer-local-map (kbd "s-v") #'yank))
#+end_src

** oxlint (JS/TS lint)
eglot diagnostics ride flymake here (=:checkers syntax= has no =-flymake=, so
=flycheck-eglot= never loads). flycheck stays free for standalone linters —
oxlint slots in there. The =oxlint= binary comes from Nix (=doom.nix=).

Gated on a =.oxlintrc.json= up the tree, so it stays silent in projects that
don't opt in. Prepended to =flycheck-checkers= so it's the auto-picked checker
in JS/TS buffers when the gate passes. Default-format output is
~file:line:col: message [Warning/rule]~.

#+begin_src emacs-lisp
(after! flycheck
  (flycheck-define-checker javascript-oxlint
    "JS/TS linter using oxlint (the oxc linter). See URL `https://oxc.rs'."
    :command ("oxlint" "--format" "default" source)
    :error-patterns
    ((warning line-start (file-name) ":" line ":" column ": "
              (message (minimal-match (one-or-more not-newline)))
              " [Warning/" (id (one-or-more (not (any "]")))) "]" line-end)
     (error   line-start (file-name) ":" line ":" column ": "
              (message (minimal-match (one-or-more not-newline)))
              " [Error/" (id (one-or-more (not (any "]")))) "]" line-end))
    :modes (web-mode js-mode js2-mode js3-mode rjsx-mode typescript-mode
            typescript-ts-mode tsx-ts-mode js-ts-mode)
    :predicate
    (lambda ()
      (locate-dominating-file (or buffer-file-name default-directory)
                              ".oxlintrc.json")))
  (add-to-list 'flycheck-checkers 'javascript-oxlint))
#+end_src

* Custom
** Session launcher (sesh replacement)
The real replacement for the tmux+sesh flow. ~SPC o s~ pops one fuzzy picker
over *everything you might want to open*:

- ~●~ workspaces already open — switch instantly (your live tmux sessions)
- ~★~ declared projects from =+vb/projects= — switch/create + spawn its servers
- ~z~ zoxide frecency dirs — open any dir you've ~cd~'d into lately
- plain projectile repos — every git project Doom knows about

…and you can just *type a path* that's in none of them. Picking a session
switches/creates its *workspace* (persp-mode), drops you in dired at the root,
and spawns declared servers as background =vterm= buffers (reached with ~SPC ,~).

- ~SPC o s~  — the fuzzy session picker (also on ~SPC p O~)
- ~SPC TAB ]~ / ~SPC TAB [~ — flip between open sessions
- ~SPC TAB 1..9~ — jump to session N (your ctrl+space+<n>)
- ~SPC ,~ — switch buffer *within* this session (the editor + each =*vterm:…*=)

To watch a server: ~SPC ,~ -> pick =*vterm:node*=. To keep it visible while you
edit: ~SPC w v~ (split), then ~SPC ,~ in the new window.

#+begin_src emacs-lisp
(defvar +vb/projects
  '(("web-builder"
     :path "~/Documents/work/web-builder"
     :servers (("docker" . "docker compose up")
               ("node"   . "pnpm run dev")))
    ("serenityOs"
     :path "~/serenityOs"
     :servers nil)
    ("server mal"
     :path "~"
     :shell ("ssh" . "ssh serenity@mal")))
  "Declared session layouts for `+vb/open-session'. Each entry:
NAME plist with :path, optional :servers (alist of NAME . SHELL-CMD run as
background vterms), optional :shell (NAME . CMD for a foreground-only shell
workspace with no editor).")

(defun +vb/vterm-ensure (name dir cmd)
  "Ensure a vterm buffer named *vterm:NAME* exists in DIR running CMD.
CMD is run as the shell's startup command (via `vterm-shell') rather than
typed in afterwards — this avoids the send-string race where input is dropped
before the PTY is ready, leaving an empty buffer. When CMD exits you drop into
a shell so the buffer stays usable. Returns the buffer; never steals the
selected window.

Uses a *login* bash (`bash -lc'): the Emacs daemon is launched by launchd/
systemd with a stripped PATH (only /usr/bin:/bin plus the nix `extraBinPackages'
from doom.nix), so a non-login shell can't find docker, pnpm, etc. A login
shell rebuilds PATH from /etc/profile + path_helper + nix. Bash (not the user's
nushell) is used for reliable POSIX `;'/`exec' semantics."
  (require 'vterm)
  (let ((bufname (format "*vterm:%s*" name)))
    (or (get-buffer bufname)
        (save-window-excursion
          (let* ((default-directory (file-name-as-directory (expand-file-name dir)))
                 (vterm-buffer-name bufname)
                 (sh (or (executable-find "bash") "/bin/bash"))
                 (vterm-shell
                  (if (and cmd (not (string-empty-p cmd)))
                      (format "%s -lc %s" sh
                              (shell-quote-argument (format "%s; exec %s -l" cmd sh)))
                    vterm-shell)))
            (vterm)
            (current-buffer))))))

(defun +vb/--open-dir (name path &optional servers)
  "Switch/create workspace NAME, open dired at PATH, spawn SERVERS as vterms.
The vterms run in the background. They are added to this workspace explicitly
via `persp-add-buffer' — buffers created off-screen (inside
`save-window-excursion') are never displayed in a workspace window, so
persp-mode never adopts them, and `SPC ,' wouldn't list them otherwise.
Reach a running server with `SPC ,'."
  (+workspace-switch name t)
  (dired path)
  (delete-other-windows)
  (dolist (s servers)
    (let ((buf (+vb/vterm-ensure (car s) path (cdr s))))
      (when (and (buffer-live-p buf) (fboundp 'persp-add-buffer))
        (persp-add-buffer buf))))
  (message "Session %s ready%s" name
           (if servers
               (format " — %d server(s) running; SPC , to view" (length servers))
             " — SPC , for its buffers")))

(defun +vb/--session-candidates ()
  "Build DISPLAY -> spec alist from open workspaces, `+vb/projects',
zoxide, and projectile. Deduped by workspace name and by absolute path."
  (let ((seen-name (make-hash-table :test 'equal))
        (seen-path (make-hash-table :test 'equal))
        (out '()))
    (cl-flet ((norm (p) (and p (directory-file-name (expand-file-name p))))
              (add  (display spec)
                (let ((n (plist-get spec :name))
                      (p (and (plist-get spec :path)
                              (directory-file-name
                               (expand-file-name (plist-get spec :path))))))
                  (unless (or (and n (gethash n seen-name))
                              (and p (gethash p seen-path)))
                    (when n (puthash n t seen-name))
                    (when p (puthash p t seen-path))
                    (push (cons display spec) out)))))
      ;; 1. open workspaces — instant switch (running sessions)
      (dolist (w (+workspace-list-names))
        (add (format "● %s" w) (list :name w :kind 'workspace)))
      ;; 2. declared projects
      (dolist (proj +vb/projects)
        (add (format "★ %-14s %s" (car proj) (or (plist-get (cdr proj) :path) ""))
             (append (list :name (car proj) :kind 'config) (cdr proj))))
      ;; 3. zoxide frecency dirs
      (when (executable-find "zoxide")
        (dolist (d (split-string
                    (shell-command-to-string "zoxide query -l") "\n" t))
          (add (format "z %s" (abbreviate-file-name d))
               (list :name (file-name-nondirectory (directory-file-name d))
                     :path d :kind 'dir))))
      ;; 4. projectile known projects
      (dolist (root (and (boundp 'projectile-known-projects)
                         projectile-known-projects))
        (add (format "  %s" (abbreviate-file-name root))
             (list :name (file-name-nondirectory (directory-file-name root))
                   :path root :kind 'dir))))
    (nreverse out)))

(defun +vb/open-session ()
  "Fuzzy session switcher — the sesh replacement. Pick a running workspace,
a declared project, a zoxide dir, a projectile repo, or type any path."
  (interactive)
  (let* ((cands  (+vb/--session-candidates))
         (choice (completing-read "Session: " (mapcar #'car cands) nil nil))
         (spec   (cdr (assoc choice cands))))
    (cond
     ;; free-typed path not in any list
     ((null spec)
      (let ((dir (expand-file-name (string-trim choice))))
        (if (file-directory-p dir)
            (+vb/--open-dir (file-name-nondirectory (directory-file-name dir)) dir)
          (user-error "No session or directory: %s" choice))))
     ;; already-open workspace
     ((eq (plist-get spec :kind) 'workspace)
      (+workspace-switch (plist-get spec :name)))
     ;; declared shell-only session (e.g. ssh mal)
     ((plist-get spec :shell)
      (+workspace-switch (plist-get spec :name) t)
      (switch-to-buffer (+vb/vterm-ensure (car (plist-get spec :shell))
                                          (plist-get spec :path)
                                          (cdr (plist-get spec :shell)))))
     ;; declared project or plain dir
     (t
      (+vb/--open-dir (plist-get spec :name)
                      (expand-file-name (plist-get spec :path))
                      (plist-get spec :servers))))))

* Android / Kotlin (Compose Multiplatform)
The closest-to-Android-Studio dev loop without the IDE. Two reload paths:

- *Fast loop (desktop):* ~SPC a h~ launches the Compose Multiplatform *desktop*
  target with JetBrains [[https://github.com/JetBrains/compose-hot-reload][Compose Hot Reload]] (~--auto~). Save any =.kt= ->
  recompile -> hot-swap into the running window, sub-second, state preserved.
  This is genuine hot reload; iterate UI here, deploy to the emulator only to
  verify the Android target. *Android Studio's Live Edit on the emulator itself
  is IDE-exclusive (JDWP code injection) and cannot be replicated in Emacs* —
  the desktop target is the supported fast path.
- *Slow loop (emulator):* ~SPC a w~ runs ~gradlew -t installDebug~ — continuous
  build that rebuilds + reinstalls the APK on every save (seconds per cycle).

Plus full emulator/device management: list & boot AVDs, list devices, logcat,
install+launch — all from Emacs, in PATH-safe vterm buffers (reused via
=+vb/vterm-ensure=; reach them with ~SPC ,~).

** Project-side prerequisites
The Emacs commands drive Gradle; the project must provide the tasks:

- *Compose Hot Reload:* apply ~org.jetbrains.compose.hot-reload~ in the desktop
  module and register a hot-run task (commonly ~hotRunJvm~). See the plugin
  README. ~SPC a h~ runs =+android-hot-run-task= (default ~:composeApp:hotRunJvm~).
- *Module/task names vary per template.* Override per-project in a
  =.dir-locals.el= at the repo root, e.g.:
  #+begin_example
  ((nil . ((+android-gradle-module . "composeApp")
           (+android-hot-run-task  . ":composeApp:hotRunJvm")
           (+android-app-id        . "com.example.app"))))
  #+end_example

** Config
SDK tools (=adb=, =emulator=) are resolved from the Android Studio SDK by
absolute path — important because the Emacs daemon runs with a stripped PATH,
and to avoid a nix-=adb= / SDK-=adb= server version clash. Override
=+android-sdk-path= if your SDK lives elsewhere.

#+begin_src emacs-lisp
(defgroup +android nil
  "Android / Compose Multiplatform development."
  :group 'tools :prefix "+android-")

(defcustom +android-sdk-path
  (or (getenv "ANDROID_HOME")
      (getenv "ANDROID_SDK_ROOT")
      (expand-file-name "~/Library/Android/sdk"))
  "Root of the Android SDK (adb under platform-tools/, emulator under emulator/)."
  :type 'directory :group '+android)

(defcustom +android-gradle-module "composeApp"
  "Gradle module holding the Android application + Compose desktop target.
Override per-project in .dir-locals.el."
  :type 'string :group '+android)

(defcustom +android-hot-run-task ":composeApp:hotRunJvm"
  "Gradle task launching the Compose *desktop* target with Compose Hot Reload.
Override per-project in .dir-locals.el. Examples: \":composeApp:hotRunJvm\",
\":desktopApp:hotRun\"."
  :type 'string :group '+android)

(defcustom +android-app-id nil
  "Application id (package) used to launch/filter the app. nil = don't auto-launch."
  :type '(choice (const nil) string) :group '+android)

;; --- SDK / project resolution ------------------------------------------
(defun +android--tool (subdir name)
  "Absolute path to SDK tool NAME under SUBDIR of `+android-sdk-path'.
Falls back to PATH; errors with a clear hint if neither resolves."
  (let ((p (expand-file-name (format "%s/%s" subdir name) +android-sdk-path)))
    (cond ((file-executable-p p) p)
          ((executable-find name))
          (t (user-error "Can't find %s — set `+android-sdk-path' (now %s)"
                         name +android-sdk-path)))))

(defun +android--adb () (+android--tool "platform-tools" "adb"))
(defun +android--emulator () (+android--tool "emulator" "emulator"))

(defun +android--root ()
  "Project root via Doom/projectile, else `default-directory'."
  (or (and (fboundp 'doom-project-root) (doom-project-root))
      (and (fboundp 'projectile-project-root) (projectile-project-root))
      default-directory))

(defun +android--gradlew ()
  "Absolute path to ./gradlew at the project root."
  (let ((gw (expand-file-name "gradlew" (+android--root))))
    (if (file-executable-p gw) gw
      (user-error "No executable ./gradlew at %s" (+android--root)))))

;; --- Emulator / device management --------------------------------------
(defun +android/start-emulator (avd)
  "Pick an AVD and boot it in a background vterm. adb sees it once booted."
  (interactive
   (list (completing-read
          "AVD: "
          (ignore-errors (process-lines (+android--emulator) "-list-avds"))
          nil t)))
  (+vb/vterm-ensure (format "emulator:%s" avd) (+android--root)
                    (format "%s -avd %s"
                            (shell-quote-argument (+android--emulator))
                            (shell-quote-argument avd)))
  (message "Booting AVD %s …" avd))

(defun +android/devices ()
  "Show attached devices/emulators (adb devices -l)."
  (interactive)
  (let ((out (shell-command-to-string
              (format "%s devices -l" (shell-quote-argument (+android--adb))))))
    (with-current-buffer (get-buffer-create "*android-devices*")
      (erase-buffer) (insert out) (goto-char (point-min))
      (display-buffer (current-buffer)))))

(defun +android/logcat ()
  "Tail colorized logcat in a vterm. Filters to `+android-app-id' if set."
  (interactive)
  (+vb/vterm-ensure "logcat" (+android--root)
                    (format "%s logcat -v color%s"
                            (shell-quote-argument (+android--adb))
                            (if +android-app-id
                                (format " | grep -F %s"
                                        (shell-quote-argument +android-app-id))
                              ""))))

;; --- Build / run loops -------------------------------------------------
(defun +android/hot-run ()
  "Launch the Compose *desktop* target with Compose Hot Reload (the fast loop).
Save .kt files -> recompile -> hot-swap into the running window. Keep it
running while you code. Requires the project's hot-run task (see
`+android-hot-run-task') and the compose-hot-reload Gradle plugin."
  (interactive)
  (+vb/vterm-ensure "compose-hot" (+android--root)
                    (format "%s %s --auto"
                            (shell-quote-argument (+android--gradlew))
                            +android-hot-run-task)))

(defun +android/install-run ()
  "Build + install the debug APK to the running device, then launch it.
Uses `compile' so Gradle/Kotlin errors are clickable. Launches via monkey when
`+android-app-id' is set."
  (interactive)
  (let ((default-directory (+android--root)))
    (compile
     (format "%s :%s:installDebug%s"
             (shell-quote-argument (+android--gradlew))
             +android-gradle-module
             (if +android-app-id
                 (format " && %s shell monkey -p %s -c android.intent.category.LAUNCHER 1"
                         (shell-quote-argument (+android--adb))
                         (shell-quote-argument +android-app-id))
               "")))))

(defun +android/watch-install ()
  "Continuous build: rebuild + reinstall to the emulator on every save (gradle -t).
The slower Android-target equivalent of hot reload. Prefer `+android/hot-run'
for fast Compose iteration."
  (interactive)
  (+vb/vterm-ensure "gradle-watch" (+android--root)
                    (format "%s -t :%s:installDebug"
                            (shell-quote-argument (+android--gradlew))
                            +android-gradle-module)))
