=emacs-lisp= block below into =config.el= on ~doom sync~. Edit this file, not
=config.el= (that one is generated — don't touch it).

To re-tangle without a full sync: ~SPC h r r~ (reload) re-tangles + reloads, or
~C-c C-v t~ (org-babel-tangle) here.

* Project launcher (sesh-style)
Replaces the tmux+sesh flow. Each project = a *workspace* (persp-mode). Opening
one switches/creates its workspace, drops you in dired at the root, and spawns
the project's long-running servers as background =vterm= buffers (reached with
~SPC ,~). Mirrors =modules/apps/cli/sesh.nix=; ~jjui~ left out on purpose.

- ~SPC p O~  — open a project layout (fuzzy)
- ~SPC TAB ]~ / ~SPC TAB [~ — flip between open project workspaces
- ~SPC TAB 1..9~ — jump to workspace N (your ctrl+space+<n>)
- ~SPC ,~ — switch buffer *within* this project (the editor + each =*vterm:…*=)

To watch a server: ~SPC ,~ -> pick =*vterm:node*=. To keep it visible while you
edit: ~SPC w v~ (split), then ~SPC ,~ in the new window.

#+begin_src emacs-lisp
(defvar +vb/projects
  '(("web-builder"
     :path "~/Documents/work/web-builder"
     :servers (("node" . "pnpm run dev")))
    ("serenityOs"
     :path "~/serenityOs"
     :servers nil)
    ("server mal"
     :path "~"
     :shell ("ssh" . "ssh serenity@mal")))
  "Project layouts for `+vb/open-project'. Each entry:
NAME plist with :path, optional :servers (alist of NAME . SHELL-CMD run as
background vterms), optional :shell (NAME . CMD for a foreground-only shell
workspace with no editor).")

(defun +vb/vterm-ensure (name dir cmd)
  "Ensure a vterm buffer named *vterm:NAME* exists in DIR running CMD.
Return the buffer. Never steals the selected window."
  (let ((bufname (format "*vterm:%s*" name)))
    (or (get-buffer bufname)
        (save-window-excursion
          (let ((default-directory (file-name-as-directory (expand-file-name dir)))
                (vterm-buffer-name bufname))
            (vterm)
            (when (and cmd (not (string-empty-p cmd)))
              (vterm-send-string cmd)
              (vterm-send-return))
            (current-buffer))))))

(defun +vb/open-project (name)
  "Open the project NAME layout: switch to its workspace, then dired + servers."
  (interactive
   (list (completing-read "Open project: " (mapcar #'car +vb/projects) nil t)))
  (let* ((spec    (cdr (assoc name +vb/projects)))
         (path    (expand-file-name (plist-get spec :path)))
         (servers (plist-get spec :servers))
         (shell   (plist-get spec :shell)))
    (+workspace-switch name t)
    (if shell
        ;; bare shell workspace (e.g. ssh to mal) — no editor, just the terminal
        (switch-to-buffer (+vb/vterm-ensure (car shell) path (cdr shell)))
      (progn
        (dired path)
        (delete-other-windows)
        (dolist (s servers)
          (+vb/vterm-ensure (car s) path (cdr s)))))
    (message "Project %s ready (SPC , to reach its buffers)" name)))

(map! :leader
      :desc "Open project layout" "p O" #'+vb/open-project)
#+end_src
